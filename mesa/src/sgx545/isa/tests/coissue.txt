# efo and fdual (ISA.md "efo", "Dual issue", "Text notation and defaults").
# The efo lines are ISA.md's corpus control words (its efo table) on r0-r3:
# the low word is the operands, the high word 0x38000000 | the control word.

efo r0, r1, r2, r3 [d=a0 i=m0/m1 a=s0+s2/s1+s2 wi0 wi1] ; 00004103 382b0600
efo r0, r1, r2, r3 [d=a0 a=s0+s2/s1+s2 wi0 wi1] ; 00004103 38230600
efo r0, r1, r2, r3 [d=i1 i=m0/m1 wi0 wi1] ; 00004103 38180600
efo r0, r1, r2, r3 [a=m0+i0/i1+m1 wi0 wi1] ; 00004103 38020600
efo r0, r1, r2, r3 [d=a0 a=m0+i0/i1+m1 wi1] ; 00004103 38220200
efo r0, r1, r2, r3 [d=a0 i=a0/m1 a=m0+s0/i1+i0 wi0 wi1 a0rneg] ; 00004103 3c2d0600
efo r0, r1, r2, r3 [d=a1 i=a0/m1 a=m0+s0/i1+i0 wi0 wi1 a0rneg] ; 00004103 3c3d0600
efo r0, r1, r2, r3 [a=m0+s0/i1+i0 m=s2*s1/s2*s2 wi0 wi1] ; 00004103 38014600
efo r0, r1, r2, r3 [d=a0] ; 00004103 38200000
efo r0, r1, r2, r3 [d=a0 i=m0/m1 a=s0+s2/s1+s2 m=s1*s0/s2*s2 wi0 wi1] ; 00004103 382b8600

# efo's i, a and m as dis.py prints them, and the enum fields at 0 written out
efo r0, r1, r2, r3 [d=a0 i=i0=m0,i1=m1 a=a0=s0+s2,a1=s1+s2 wi0 wi1] == efo r0, r1, r2, r3 [d=a0 i=m0/m1 a=s0+s2/s1+s2 wi0 wi1]
efo r0, r1, r2, r3 [d=i0 i=i0=a0,i1=a1 a=a0=m0+m1,a1=i1+i0 m=m0=s0*s2,m1=s1*s2] == efo r0, r1, r2, r3
efo r0, r1, r2, r3 [m=m0=s1*s0,m1=s2*s2] == efo r0, r1, r2, r3 [m=s1*s0/s2*s2]
efo r0, r1, r2, r3 [d=2 i=2 a=3 m=0] == efo r0, r1, r2, r3 [d=a0 i=m0/m1 a=s0+s2/s1+s2]
